cmake_minimum_required(VERSION 3.21)

if(WIN32 AND NOT DEFINED CMAKE_MSVC_RUNTIME_LIBRARY)
    set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL")
endif()

project(mimium_clap_plugin_wrappers LANGUAGES C CXX)

if(APPLE)
    enable_language(OBJC)
    enable_language(OBJCXX)
endif()

if(NOT DEFINED CMAKE_CXX_STANDARD)
    set(CMAKE_CXX_STANDARD 23)
endif()

set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

if(NOT DEFINED CLAP_WRAPPER_ROOT)
    message(FATAL_ERROR "CLAP_WRAPPER_ROOT must point to the clap-wrapper submodule")
endif()

if(NOT DEFINED CLAP_PLUGIN_BUNDLE)
    message(FATAL_ERROR "CLAP_PLUGIN_BUNDLE must point to a staged .clap bundle")
endif()

if(NOT EXISTS "${CLAP_PLUGIN_BUNDLE}")
    message(FATAL_ERROR "CLAP_PLUGIN_BUNDLE does not exist: ${CLAP_PLUGIN_BUNDLE}")
endif()

if(NOT DEFINED CLAP_WRAPPER_OUTPUT_NAME)
    set(CLAP_WRAPPER_OUTPUT_NAME "Mimium CLAP Plugin")
endif()

if(NOT DEFINED CLAP_WRAPPER_BUNDLE_IDENTIFIER)
    set(CLAP_WRAPPER_BUNDLE_IDENTIFIER "org.mimium.mimium-clap-plugin")
endif()

if(NOT DEFINED CLAP_WRAPPER_BUNDLE_VERSION)
    set(CLAP_WRAPPER_BUNDLE_VERSION "0.1.0")
endif()

option(CLAP_WRAPPER_BUILD_AUV2 "Build the AUv2 wrapper target" OFF)

add_subdirectory("${CLAP_WRAPPER_ROOT}" "${CMAKE_BINARY_DIR}/clap-wrapper" EXCLUDE_FROM_ALL)

add_library(mimium_clap_plugin_vst3 MODULE)
target_add_vst3_wrapper(
    TARGET mimium_clap_plugin_vst3
    OUTPUT_NAME "${CLAP_WRAPPER_OUTPUT_NAME}"
    BUNDLE_IDENTIFIER "${CLAP_WRAPPER_BUNDLE_IDENTIFIER}.vst3"
    BUNDLE_VERSION "${CLAP_WRAPPER_BUNDLE_VERSION}"
    MACOS_EMBEDDED_CLAP_LOCATION "${CLAP_PLUGIN_BUNDLE}"
)

if(APPLE AND CLAP_WRAPPER_BUILD_AUV2)
    if(NOT DEFINED CLAP_WRAPPER_AUV2_MANUFACTURER_NAME)
        message(FATAL_ERROR "CLAP_WRAPPER_AUV2_MANUFACTURER_NAME is required when CLAP_WRAPPER_BUILD_AUV2=ON")
    endif()

    if(NOT DEFINED CLAP_WRAPPER_AUV2_MANUFACTURER_CODE)
        message(FATAL_ERROR "CLAP_WRAPPER_AUV2_MANUFACTURER_CODE is required when CLAP_WRAPPER_BUILD_AUV2=ON")
    endif()

    if(NOT DEFINED CLAP_WRAPPER_AUV2_SUBTYPE_CODE)
        message(FATAL_ERROR "CLAP_WRAPPER_AUV2_SUBTYPE_CODE is required when CLAP_WRAPPER_BUILD_AUV2=ON")
    endif()

    if(NOT DEFINED CLAP_WRAPPER_AUV2_INSTRUMENT_TYPE)
        message(FATAL_ERROR "CLAP_WRAPPER_AUV2_INSTRUMENT_TYPE is required when CLAP_WRAPPER_BUILD_AUV2=ON")
    endif()

    add_library(mimium_clap_plugin_auv2 MODULE)
    target_add_auv2_wrapper(
        TARGET mimium_clap_plugin_auv2
        OUTPUT_NAME "${CLAP_WRAPPER_OUTPUT_NAME}"
        BUNDLE_IDENTIFIER "${CLAP_WRAPPER_BUNDLE_IDENTIFIER}.auv2"
        BUNDLE_VERSION "${CLAP_WRAPPER_BUNDLE_VERSION}"
        MANUFACTURER_NAME "${CLAP_WRAPPER_AUV2_MANUFACTURER_NAME}"
        MANUFACTURER_CODE "${CLAP_WRAPPER_AUV2_MANUFACTURER_CODE}"
        SUBTYPE_CODE "${CLAP_WRAPPER_AUV2_SUBTYPE_CODE}"
        INSTRUMENT_TYPE "${CLAP_WRAPPER_AUV2_INSTRUMENT_TYPE}"
        MACOS_EMBEDDED_CLAP_LOCATION "${CLAP_PLUGIN_BUNDLE}"
    )
endif()
